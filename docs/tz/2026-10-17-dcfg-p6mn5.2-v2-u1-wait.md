```bash
# Final delivery state, pushed to the caller by inotify events instead of agent polling.
do_wait(){
	local atom=$1 t=$2 end left line
	[[ $t =~ ^[1-9][0-9]*$ ]] || usage_err "invalid timeout: $t"
	end=$(( $(date +%s) + t ))
	while :; do
		line=$(do_status "$atom") || return 2
		case $line in
			*' delivered'*|*' wait-owner'*|*' prod-hold'*) echo "$line"; return 0 ;;
			*' parked '*|*' failed '*) echo "$line"; return 1 ;;
		esac
		left=$(( end - $(date +%s) ))
		(( left > 0 )) || { echo "$line"; return 3; }
		# cap 30 s: covers an event fired between do_status and inotifywait start
		inotifywait -qq -r -t $(( left < 30 ? left : 30 )) -e close_write,moved_to,create,delete "$SHIP_POST_STATE_DIR" >/dev/null 2>&1
	done
}
```
